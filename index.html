datiInvestimenti.raw.sort((a, b) => a.data.localeCompare(b.data)); datiInvestimenti.raw.forEach(r => { cumulativo += r.imp; r.patrimonio = cumulativo; }); if (datiInvestimenti.raw.length > 0) { datiInvestimenti.patrimonioIniziale = datiInvestimenti.raw[0].patrimonio; datiInvestimenti.patrimonioFinale = cumulativo; } } console.log(`Elaborati: ${elaborati}, Saltati: ${saltati}`); console.log(`Movimenti: ${datiInvestimenti.movimenti.length}, Titoli: ${Object.keys(datiInvestimenti.titoli).length}`); console.log(`Patrimonio: ${datiInvestimenti.patrimonioIniziale} -> ${datiInvestimenti.patrimonioFinale}`); return elaborati > 0; } // Gestisce il caricamento file investimenti document.getElementById('investimentiFileInput').addEventListener('change', function(e) { const file = e.target.files[0];
